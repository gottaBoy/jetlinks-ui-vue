### 2. model/api.ts (API接口定义)

```typescript
import { request } from '@jetlinks/ui-vue';

export const parallelDrivingApi = {
  // 绑定驾驶舱到车端
  bind(params: { cockpitDeviceId: string; vehicleDeviceId: string }) {
    return request.post('/api/parallel-driving/bind', null, { params });
  },

  // 解绑
  unbind(params: { cockpitDeviceId: string; vehicleDeviceId: string }) {
    return request.delete('/api/parallel-driving/unbind', { params });
  },

  // 查询驾驶舱绑定的车
  getBoundVehicle(cockpitId: string) {
    return request.get(`/api/parallel-driving/cockpit/${cockpitId}/vehicle`);
  },

  // 查询车被哪个驾驶舱绑定
  getBoundCockpit(vehicleId: string) {
    return request.get(`/api/parallel-driving/vehicle/${vehicleId}/cockpit`);
  },

  // 检查控制权限
  checkPermission(params: { cockpitDeviceId: string; vehicleDeviceId: string }) {
    return request.get('/api/parallel-driving/permission/check', { params });
  },
};
```

### 4. components/BindForm.vue (绑定表单组件)

```vue
<template>
  <a-form :model="form" @submit="handleSubmit">
    <a-form-item label="驾驶舱设备" required>
      <a-select v-model="form.cockpitDeviceId" placeholder="请选择驾驶舱设备">
        <a-option v-for="device in cockpitDevices" :key="device.id" :value="device.id">
          {{ device.name }}
        </a-option>
      </a-select>
    </a-form-item>
    
    <a-form-item label="车端设备" required>
      <a-select v-model="form.vehicleDeviceId" placeholder="请选择车端设备">
        <a-option v-for="device in vehicleDevices" :key="device.id" :value="device.id">
          {{ device.name }}
        </a-option>
      </a-select>
    </a-form-item>
    
    <a-form-item>
      <a-button type="primary" html-type="submit" :loading="loading">
        绑定
      </a-button>
    </a-form-item>
  </a-form>
</template>

<script setup lang="ts">
import { ref, reactive } from 'vue';
import { message } from '@jetlinks/ui-vue';
import { parallelDrivingApi } from '../model/api';

const emit = defineEmits(['success']);

const loading = ref(false);
const form = reactive({
  cockpitDeviceId: '',
  vehicleDeviceId: '',
});

const cockpitDevices = ref([]); // 从设备管理模块获取驾驶舱设备列表
const vehicleDevices = ref([]); // 从设备管理模块获取车端设备列表

const handleSubmit = async () => {
  if (!form.cockpitDeviceId || !form.vehicleDeviceId) {
    message.error('请选择驾驶舱和车端设备');
    return;
  }

  loading.value = true;
  try {
    await parallelDrivingApi.bind({
      cockpitDeviceId: form.cockpitDeviceId,
      vehicleDeviceId: form.vehicleDeviceId,
    });
    message.success('绑定成功');
    emit('success');
    // 重置表单
    form.cockpitDeviceId = '';
    form.vehicleDeviceId = '';
  } catch (error) {
    message.error(error.message || '绑定失败');
  } finally {
    loading.value = false;
  }
};
</script>
```
